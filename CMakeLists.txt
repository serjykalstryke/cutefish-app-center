cmake_minimum_required(VERSION 3.20)
project(cutefish-app-center VERSION 0.2.3 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(CutefishToolkit REQUIRED)
add_executable(cutefish-app-center src/main.cpp)
target_compile_features(cutefish-app-center PRIVATE cxx_std_20)
target_link_libraries(cutefish-app-center PRIVATE Cutefish::Toolkit Cutefish::ToolkitBase)
target_compile_options(cutefish-app-center PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(cutefish-app-center PRIVATE APP_CENTER_VERSION="${PROJECT_VERSION}" APP_CENTER_BRANDING_FILE="${CMAKE_INSTALL_FULL_DATADIR}/cutefish-app-center/branding.json" APP_CENTER_BACKEND="${CMAKE_INSTALL_FULL_LIBEXECDIR}/cutefish-app-center/app_center_backend.py")
install(TARGETS cutefish-app-center RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS backend/app_center_backend.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/cutefish-app-center)
install(FILES data/org.cutefish.AppCenter.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/org.cutefish.AppCenter.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES data/org.cutefish.AppCenter.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
option(APP_CENTER_BUILD_NATIVE_TEST "Build the native headless integration test" OFF)
if(APP_CENTER_BUILD_NATIVE_TEST)
    add_executable(cutefish-app-center-icons-smoke tests/native_icons.cpp)
    target_compile_features(cutefish-app-center-icons-smoke PRIVATE cxx_std_20)
    target_compile_options(cutefish-app-center-icons-smoke PRIVATE -UNDEBUG)
    target_link_libraries(cutefish-app-center-icons-smoke PRIVATE Cutefish::Toolkit Cutefish::ToolkitBase)
    add_executable(cutefish-app-center-home-smoke tests/native_home.cpp)
    target_compile_features(cutefish-app-center-home-smoke PRIVATE cxx_std_20)
    target_compile_options(cutefish-app-center-home-smoke PRIVATE -UNDEBUG)
    target_link_libraries(cutefish-app-center-home-smoke PRIVATE Cutefish::Toolkit Cutefish::ToolkitBase)
    add_executable(cutefish-app-center-details-smoke tests/native_details.cpp)
    target_compile_features(cutefish-app-center-details-smoke PRIVATE cxx_std_20)
    target_compile_options(cutefish-app-center-details-smoke PRIVATE -UNDEBUG)
    target_link_libraries(cutefish-app-center-details-smoke PRIVATE Cutefish::Toolkit Cutefish::ToolkitBase)
    add_executable(cutefish-app-center-native-smoke tests/native_smoke.cpp)
    target_compile_features(cutefish-app-center-native-smoke PRIVATE cxx_std_20)
    target_compile_options(cutefish-app-center-native-smoke PRIVATE -UNDEBUG)
    target_link_libraries(cutefish-app-center-native-smoke PRIVATE Cutefish::Toolkit Cutefish::ToolkitBase)
endif()

install(FILES data/branding.json DESTINATION ${CMAKE_INSTALL_DATADIR}/cutefish-app-center)
