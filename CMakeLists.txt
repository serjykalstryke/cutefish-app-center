cmake_minimum_required(VERSION 3.20)
project(cutefish-store VERSION 0.1.1 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(CutefishToolkit REQUIRED)
add_executable(cutefish-store src/main.cpp)
target_compile_features(cutefish-store PRIVATE cxx_std_20)
target_link_libraries(cutefish-store PRIVATE Cutefish::Toolkit Cutefish::ToolkitBase)
target_compile_options(cutefish-store PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(cutefish-store PRIVATE STORE_BRANDING_FILE="${CMAKE_INSTALL_FULL_DATADIR}/cutefish-store/branding.json" STORE_BACKEND="${CMAKE_INSTALL_FULL_LIBEXECDIR}/cutefish-store/store_backend.py")
install(TARGETS cutefish-store RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS backend/store_backend.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/cutefish-store)
install(FILES data/org.cutefish.Store.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/org.cutefish.Store.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES data/org.cutefish.Store.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
option(STORE_BUILD_NATIVE_TEST "Build the native headless integration test" OFF)
if(STORE_BUILD_NATIVE_TEST)
    add_executable(cutefish-store-native-smoke tests/native_smoke.cpp)
    target_compile_features(cutefish-store-native-smoke PRIVATE cxx_std_20)
    target_link_libraries(cutefish-store-native-smoke PRIVATE Cutefish::Toolkit Cutefish::ToolkitBase)
endif()

install(FILES data/branding.json DESTINATION ${CMAKE_INSTALL_DATADIR}/cutefish-store)
